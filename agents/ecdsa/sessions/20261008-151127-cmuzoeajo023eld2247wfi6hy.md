import sys, json, random

def run(p, a, b, G, P, n, mask, W):
    rnd = random.Random()
    def add(A, B):
        if A is None: return B
        if B is None: return A
        if A[0] == B[0]:
            if (A[1] + B[1]) % p == 0: return None
            l = (3*A[0]*A[0] + a) * pow(2*A[1], -1, p) % p
        else:
            l = (B[1]-A[1]) * pow(B[0]-A[0], -1, p) % p
        x = (l*l - A[0] - B[0]) % p
        return (x, (l*(A[0]-x) - A[1]) % p)
    def mul(k, A):
        R = None
        while k:
            if k & 1: R = add(R, A)
            A = add(A, A); k >>= 1
        return R
    S = 64
    tx = []; ty = []; ta = []; tb = []
    for _ in range(S):
        u = rnd.randrange(1, n); v = rnd.randrange(1, n)
        Q = add(mul(u, G), mul(v, P))
        tx.append(Q[0]); ty.append(Q[1]); ta.append(u); tb.append(v)
    X = []; Y = []; A = []; B = []
    for _ in range(W):
        u = rnd.randrange(1, n); v = rnd.randrange(1, n)
        Q = add(mul(u, G), mul(v, P))
        X.append(Q[0]); Y.append(Q[1]); A.append(u); B.append(v)
    rng = range(W)
    rrng = range(W-1, -1, -1)
    seen = {}
    while True:
        J = [x & 63 for x in X]
        D = [(X[i] - tx[J[i]]) % p or 1 for i in rng]
        pre = [0]*W
        acc = 1
        for i in rng:
            acc = acc * D[i] % p
            pre[i] = acc
        inv = pow(acc, -1, p)
        for i in rrng:
            di = inv * pre[i-1] % p if i else inv
            inv = inv * D[i] % p
            j = J[i]
            x1 = X[i]; y1 = Y[i]
            l = (y1 - ty[j]) * di % p
            x3 = (l*l - x1 - tx[j]) % p
            Y[i] = (l*(x1 - x3) - y1) % p
            X[i] = x3
            u = A[i] = (A[i] + ta[j]) % n
            v = B[i] = (B[i] + tb[j]) % n
            if (x3 & mask) == 0:

                    if dv:
                        try:
                            return du * pow(dv, -1, n) % n
                        except ValueError:
                            pass
                else:
                    seen[x3] = (y, u, v)
