import json
import math

def solve():

    Gy = int(data["Gy"])

        if P1 is None:
            return P2
        if P2 is None:
            return P1
        x1, y1 = P1
        x2, y2 = P2

                return None

            sl = (3 * x1 * x1 + a) * inv % p
        else:
            inv = pow(x2 - x1, -1, p)
            sl = (y2 - y1) * inv % p
        x3 = (sl * sl - x1 - x2) % p
        y3 = (sl * (x1 - x3) - y1) % p
        return (x3, y3)

    def point_mul(k, P):
        R = None
        T = P
        while k > 0:
            if k & 1:
                R = point_add(R, T)
            T = point_add(T, T)
            k >>= 1
        return R
