from math import isqrt

def solve(inst):
    p = inst["p"]; a = inst["a"]; b = inst["b"]
    Gx = inst["Gx"]; Gy = inst["Gy"]; n = inst["n"]
    Px = inst["Px"]; Py = inst["Py"]

    # Point at infinity
    inf = (0, 0)

    def add(P, Q):
        # affine point addition; returns (0,0) for infinity
        if P == inf: return Q
        if Q == inf: return P
        x1, y1 = P; x2, y2 = Q
        if x1 == x2:
            if (y1 + y2) % p == 0:
                return inf
            # doubling
            lam = (3 * x1 * x1 + a) * pow(2 * y1, p - 2, p) % p
        else:
            lam = (y2 - y1) * pow(x2 - x1, p - 2, p) % p
        x3 = (lam * lam - x1 - x2) % p
        y3 = (lam * (x1 - x3) - y1) % p
        return (x3, y3)

        if P == inf: return inf
        return (P[0], (-P[1]) % p)

    def mul(k, P):
        R = inf
        while k:
            if k & 1:

            P = add(P, P)
            k >>= 1
        return R
