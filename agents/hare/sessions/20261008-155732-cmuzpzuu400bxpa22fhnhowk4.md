#!/usr/bin/env python3
"""48-bit ECDLP: negation-map BSGS, packed open-addressed table.

k = i*M + j or i*M - j with M = 2m+1, m = isqrt(n)//2+1.
Baby x-coordinates live in an array('Q') (store x+1, 0 = empty) so a
full 48-bit order (~8.4e6 babies, 128MiB) does not blow a Python dict.
"""
import json
import sys
from array import array
from math import isqrt


def solve(p, a, Gx, Gy, n, Px, Py):
    if Px == 0 and Py == 0:
        return 0

    Gx %= p
    Gy %= p
    Px %= p
    Py %= p

    m = isqrt(n) // 2 + 1
    M = 2 * m + 1
    BATCH = 128

    slots = 1
    need = (m * 3) // 2 + 8
    while slots < need:
        slots <<= 1
    mask = slots - 1

    Xs = array("Q", [0]) * slots
    Js = array("I", [0]) * slots
    Xg = Xs.__getitem__
    Xs_set = Xs.__setitem__
    Jg = Js.__getitem__
    Js_set = Js.__setitem__

    def inv_batch(zs, L):
        pref = [0] * L
        acc = zs[0]
        pref[0] = acc
        for i in range(1, L):
            acc = (acc * zs[i]) % p
            pref[i] = acc
        iv = pow(acc, -1, p)
        out = [0] * L
        for i in range(L - 1, 0, -1):
            out[i] = (iv * pref[i - 1]) % p
            iv = (iv * zs[i]) % p
        out[0] = iv
        return out

    def put(x, val):
        h = x & mask
        while True:
            if Xg(h) == 0:
                Xs_set(h, x + 1)
                Js_set(h, val)
                return
            if Xg(h) == x + 1:
                return
            h = (h + 1) & mask

    def get(x):
        h = x & mask
        key = x + 1
        while True:
            xv = Xg(h)
            if xv == 0:
                return None
            if xv == key:
                return Jg(h)
            h = (h + 1) & mask

    # S[i] = i*G for i = 1..BATCH
    Sx = [0] * (BATCH + 1)
    Sy = [0] * (BATCH + 1)
    Sx[1], Sy[1] = Gx, Gy
    inv2 = pow(2, -1, p)
    lam = ((3 * Gx * Gx + a) * ((2 * Gy) % p * inv2 % p)) % p
    # simpler doubling
    lam = ((3 * Gx * Gx + a) * pow((2 * Gy) % p, -1, p)) % p
    nx = (lam * lam - 2 * Gx) % p
    ny = (lam * (Gx - nx) - Gy) % p
    Sx[2], Sy[2] = nx, ny
    for i in range(3, BATCH + 1):
        dx = (Gx - Sx[i - 1]) % p
        lam = ((Gy - Sy[i - 1]) * pow(dx, -1, p)) % p
        nx = (lam * lam - Sx[i - 1] - Gx) % p
        ny = (lam * (Sx[i - 1] - nx) - Sy[i - 1]) % p
        Sx[i], Sy[i] = nx, ny

    limit = BATCH if m > BATCH else m
    for i in range(1, limit + 1):
        put(Sx[i], (i << 1) | (Sy[i] & 1))

    rx, ry = Sx[limit], Sy[limit]
    j = limit
    while j < m:
        cnt = BATCH if j + BATCH <= m else m - j
        dxs = [1] * cnt
        dbl = 0
        for t in range(cnt):
            dx = (Sx[t + 1] - rx) % p
            if dx == 0:
                dbl |= 1 << t
            else:
                dxs[t] = dx
        ivs = inv_batch(dxs, cnt)
        for t in range(cnt):
            i = t + 1
            if (dbl >> t) & 1:
                lam = ((3 * rx * rx + a) * pow((2 * ry) % p, -1, p)) % p
                nx = (lam * lam - 2 * rx) % p
                ny = (lam * (rx - nx) - ry) % p
            else:
                lam = ((Sy[i] - ry) * ivs[t]) % p
                nx = (lam * lam - rx - Sx[i]) % p
                ny = (lam * (rx - nx) - ry) % p
            put(nx, ((j + i) << 1) | (ny & 1))
            if i == cnt:
                rx, ry = nx, ny
        j += cnt

    # D = M*G, U[i] = -(i*D)
    ax, ay = Gx, Gy
    rx_, ry_ = None, None
    kk = M
    while kk:
        if kk & 1:
            if rx_ is None:
                rx_, ry_ = ax, ay
            else:
                dx = (ax - rx_) % p
                if dx == 0:
                    if (ay + ry_) % p == 0:
                        rx_, ry_ = None, None
                    else:
                        lam = ((3 * ax * ax + a) * pow((2 * ay) % p, -1, p)) % p
                        nx = (lam * lam - 2 * ax) % p
                        ny = (lam * (ax - nx) - ay) % p
                        rx_, ry_ = nx, ny
                else:
                    lam = ((ay - ry_) * pow(dx, -1, p)) % p
                    nx = (lam * lam - rx_ - ax) % p
                    ny = (lam * (rx_ - nx) - ry_) % p
                    rx_, ry_ = nx, ny
        lam = ((3 * ax * ax + a) * pow((2 * ay) % p, -1, p)) % p
        nx = (lam * lam - 2 * ax) % p
        ny = (lam * (ax - nx) - ay) % p
        ax, ay = nx, ny
        kk >>= 1

    if rx_ is None:
        # M*G = infinity => M divisible by n, impossible for M < n
        return 0

    Ux = [0] * (BATCH + 1)
    Uy = [0] * (BATCH + 1)
    Ux[1], Uy[1] = rx_, (-ry_) % p
    lam = ((3 * Ux[1] * Ux[1] + a) * pow((2 * Uy[1]) % p, -1, p)) % p
    nx = (lam * lam - 2 * Ux[1]) % p
    ny = (lam * (Ux[1] - nx) - Uy[1]) % p
    Ux[2], Uy[2] = nx, ny
    ux1, uy1 = Ux[1], Uy[1]
    for i in range(3, BATCH + 1):
        dx = (ux1 - Ux[i - 1]) % p
        if dx == 0:
            lam = ((3 * ux1 * ux1 + a) * pow((2 * uy1) % p, -1, p)) % p
            nx = (lam * lam - 2 * ux1) % p
            ny = (lam * (ux1 - nx) - uy1) % p
        else:
            lam = ((uy1 - Uy[i - 1]) * pow(dx, -1, p)) % p
            nx = (lam * lam - Ux[i - 1] - ux1) % p
            ny = (lam * (Ux[i - 1] - nx) - Uy[i - 1]) % p
        Ux[i], Uy[i] = nx, ny

    # initial lookup: i = 0
    hit = get(Px)
    if hit is not None:
        jv = hit >> 1
        if (Py & 1) == (hit & 1):
            return jv % n
        return (n - jv) % n

    qx, qy = Px, Py
    max_i = (n - 1 + m) // M + 1
    i_base = 0
    while i_base < max_i:
        cnt = BATCH if i_base + BATCH <= max_i else max_i - i_base
        if cnt <= 0:
            break
        dxs = [1] * cnt
        dbl = 0
        infm = 0
        for t in range(cnt):
            dx = (Ux[t + 1] - qx) % p
            if dx == 0:
                if (qy + Uy[t + 1]) % p == 0:
                    infm |= 1 << t
                else:
                    dbl |= 1 << t
            else:
                dxs[t] = dx
        ivs = inv_batch(dxs, cnt)
        for t in range(cnt):
            i = t + 1
            idx = i_base + i
            if (infm >> t) & 1:
                return (idx * M) % n
            ux = Ux[i]
            uy = Uy[i]
            if (dbl >> t) & 1:
                lam = ((3 * qx * qx + a) * pow((2 * qy) % p, -1, p)) % p
                nx = (lam * lam - 2 * qx) % p
                ny = (lam * (qx - nx) - qy) % p
            else:
                lam = ((uy - qy) * ivs[t]) % p
                nx = (lam * lam - qx - ux) % p
                ny = (lam * (qx - nx) - qy) % p
            hit = get(nx)
            if hit is not None:
                jv = hit >> 1
                if (ny & 1) == (hit & 1):
                    return (idx * M + jv) % n
                return (idx * M - jv) % n
            if i == cnt:
                qx, qy = nx, ny
        i_base += cnt
    return 0


def main():
    data = json.loads(sys.stdin.read())
    k = solve(
        int(data["p"]),
        int(data["a"]),
        int(data["Gx"]),
        int(data["Gy"]),
        int(data["n"]),
        int(data["Px"]),
        int(data["Py"]),
    )
    print(f"k={k}")


if __name__ == "__main__":
    main()
