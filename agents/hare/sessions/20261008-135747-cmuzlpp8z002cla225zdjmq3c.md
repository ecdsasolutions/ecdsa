#!/usr/bin/env python3
"""ECDLP baby-step giant-step. Affine + extended-gcd inverses."""
import json
import sys


def solve(p, a, Gx, Gy, n, Px, Py):
    # identity encoding used by some instances
    if Px == 0 and Py == 0:
        return 0

    m = int(n**0.5) + 1
    inv = lambda x: pow(x, -1, p)

    baby = {}
    x, y = Gx, Gy
    baby[x] = (1, y)
    # walk j = 2 .. m-1 by adding G. First step is a doubling.
    if y == 0:
        # 2G = infinity, order 2 — not expected for prime order > 2
        x = y = None
    else:
        lam = ((3 * x * x + a) * inv((2 * y) % p)) % p
        nx = (lam * lam - 2 * x) % p
        ny = (lam * (x - nx) - y) % p
        x, y = nx, ny
        baby[x] = (2, y)

    # subsequent adds are never doublings (order > 2)
    two = 2
    for j in range(3, m):
        dx = (Gx - x) % p
        lam = ((Gy - y) * inv(dx)) % p
        nx = (lam * lam - x - Gx) % p
        ny = (lam * (x - nx) - y) % p
        x, y = nx, ny
        baby[x] = (j, y)

    # m*G by double-and-add
    def add(x1, y1, x2, y2):
        if x1 is None:
            return x2, y2
        if x2 is None:
            return x1, y1
        if x1 == x2:
            s = (y1 + y2) % p
            if s == 0:
                return None, None
            lam = ((3 * x1 * x1 + a) * inv((2 * y1) % p)) % p
            nx = (lam * lam - 2 * x1) % p
            ny = (lam * (x1 - nx) - y1) % p
            return nx, ny
        lam = ((y2 - y1) * inv((x2 - x1) % p)) % p
        nx = (lam * lam - x1 - x2) % p
        ny = (lam * (x1 - nx) - y1) % p
        return nx, ny

    def mul(k):
        rx, ry = None, None
        ax, ay = Gx, Gy
        while k:
            if k & 1:
                rx, ry = add(rx, ry, ax, ay)
            ax, ay = add(ax, ay, ax, ay)
            k >>= 1
        return rx, ry

    mx, my = mul(m)
    # -mG
    gx, gy = mx, (-my) % p

    qx, qy = Px, Py
    # k = i*m + j, j in 1..m-1 or we also need j=0 (i*m)*G = Q
    # baby has j=1..m-1. j=0 means Q is infinity, i.e. P = i*m*G, k=i*m.
    for i in range(m + 1):
        if qx is None:
            return (i * m) % n
        hit = baby.get(qx)
        if hit is not None and hit[1] == qy:
            return (i * m + hit[0]) % n
        # Q := Q + (-mG)
        if qx == gx:
            s = (qy + gy) % p
            if s == 0:
                qx, qy = None, None
                continue
            lam = ((3 * qx * qx + a) * inv((2 * qy) % p)) % p
            nx = (lam * lam - 2 * qx) % p
            ny = (lam * (qx - nx) - qy) % p
            qx, qy = nx, ny
        else:
            lam = ((gy - qy) * inv((gx - qx) % p)) % p
            nx = (lam * lam - qx - gx) % p
            ny = (lam * (qx - nx) - qy) % p
            qx, qy = nx, ny
    return None


def main():
    data = json.loads(sys.stdin.read())
    k = solve(
        int(data["p"]),
        int(data["a"]),
        int(data["Gx"]),
        int(data["Gy"]),
        int(data["n"]),
        int(data["Px"]),
        int(data["Py"]),
    )
    print(f"k={0 if k is None else k}")


if __name__ == "__main__":
    main()
