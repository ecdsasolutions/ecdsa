#!/usr/bin/env python3
"""Negation-map BSGS with Montgomery batch inversion for ECDLP."""
import json
import sys


def isqrt(x):
    r = int(x ** 0.5)
    while r * r > x:
        r -= 1
    while (r + 1) * (r + 1) <= x:
        r += 1
    return r


def solve(p, a, Gx, Gy, n, Px, Py):
    if Px == 0 and Py == 0:
        return 0

    # Balanced worst case: ~sqrt(2n) group ops, covers Z/n via stride 2m+1.
    m = isqrt(n // 2) + 1
    stride = 2 * m + 1
    BATCH = 64

    def inv(x):
        return pow(x, -1, p)

    def add(x1, y1, x2, y2):
        if x1 is None:
            return x2, y2
        if x2 is None:
            return x1, y1
        if x1 == x2:
            if (y1 + y2) % p == 0:
                return None, None
            lam = (3 * x1 * x1 + a) * inv((2 * y1) % p) % p
            nx = (lam * lam - 2 * x1) % p
            ny = (lam * (x1 - nx) - y1) % p
            return nx, ny
        lam = (y2 - y1) * inv((x2 - x1) % p) % p
        nx = (lam * lam - x1 - x2) % p
        ny = (lam * (x1 - nx) - y1) % p
        return nx, ny

    def mul(k):
        rx = ry = None
        ax, ay = Gx, Gy
        while k:
            if k & 1:
                rx, ry = add(rx, ry, ax, ay)
            ax, ay = add(ax, ay, ax, ay)
            k >>= 1
        return rx, ry

    def batch_add(xs, ys, tx, ty, active):
        """Add (tx,ty) to each active point in place. None means infinity."""
        idxs = []
        ds = []
        for i in active:
            x1 = xs[i]
            if x1 is None:
                continue
            d = (tx - x1) % p
            if d == 0:
                y1 = ys[i]
                if (y1 + ty) % p == 0:
                    xs[i] = None
                    ys[i] = None
                else:
                    lam = (3 * x1 * x1 + a) * inv((2 * y1) % p) % p
                    nx = (lam * lam - 2 * x1) % p
                    ys[i] = (lam * (x1 - nx) - y1) % p
                    xs[i] = nx
            else:
                idxs.append(i)
                ds.append(d)
        L = len(ds)
        if L == 0:
            return
        pref = [0] * L
        pref[0] = ds[0]
        for i in range(1, L):
            pref[i] = (pref[i - 1] * ds[i]) % p
        inv_all = inv(pref[-1])
        invs = [0] * L
        for i in range(L - 1, 0, -1):
            invs[i] = (pref[i - 1] * inv_all) % p
            inv_all = (inv_all * ds[i]) % p
        invs[0] = inv_all
        for t in range(L):
            i = idxs[t]
            x1 = xs[i]
            y1 = ys[i]
            lam = ((ty - y1) * invs[t]) % p
            nx = (lam * lam - x1 - tx) % p
            ys[i] = (lam * (x1 - nx) - y1) % p
            xs[i] = nx

    # Baby steps: lanes j=1..B hold jG, then add B*G.
    # Store x -> (j << 1) | (y & 1). p odd => parity separates y and -y.
    baby = {}
    nlanes = BATCH if m >= BATCH else m
    xs = [None] * nlanes
    ys = [None] * nlanes
    # sequential first nlanes multiples
    x, y = Gx, Gy
    xs[0], ys[0] = x, y
    baby[x] = (1 << 1) | (y & 1)
    for j in range(1, nlanes):
        x, y = add(x, y, Gx, Gy)
        xs[j], ys[j] = x, y
        baby[x] = ((j + 1) << 1) | (y & 1)

    if m > nlanes:
        step = mul(nlanes)  # nlanes * G
        sx, sy = step
        active = list(range(nlanes))
        rounds = (m - 1) // nlanes  # additional steps per lane
        base = nlanes
        for r in range(rounds):
            batch_add(xs, ys, sx, sy, active)
            jj = base + r * nlanes
            for lane in range(nlanes):
                j = jj + lane + 1
                if j > m:
                    break
                baby[xs[lane]] = (j << 1) | (ys[lane] & 1)

    # Giant steps: Q_i = P - i*stride*G, k = i*stride ± j
    Mx, My = mul(stride)
    nMx, nMy = Mx, (-My) % p  # -stride*G
    max_i = (n // stride) + 2

    # direct check of P (i=0) and walk
    # Use batched lanes of giant steps.
    gxs = [None] * nlanes
    gys = [None] * nlanes
    gxs[0], gys[0] = Px, Py
    cx, cy = Px, Py
    for lane in range(1, nlanes):
        cx, cy = add(cx, cy, nMx, nMy)
        gxs[lane], gys[lane] = cx, cy

    # step between lanes of one batch: nlanes * (-stride G)
    bsx, bsy = mul(nlanes)
    # bsx is nlanes*G; we need nlanes*(-stride*G) = -nlanes * stride * G
    # easier: negate nlanes * stride * G
    bsx, bsy = mul(nlanes * stride)
    bsx, bsy = bsx, (-bsy) % p

    get = baby.get
    i0 = 0
    while i0 <= max_i:
        for lane in range(nlanes):
            i = i0 + lane
            x1 = gxs[lane]
            if x1 is None:
                if i <= max_i:
                    return (i * stride) % n
                continue
            hit = get(x1)
            if hit is not None:
                j = hit >> 1
                if (gys[lane] & 1) == (hit & 1):
                    return (i * stride + j) % n
                return (i * stride - j) % n
        if i0 + nlanes > max_i:
            break
        batch_add(gxs, gys, bsx, bsy, range(nlanes))
        i0 += nlanes
    return None


def main():
    data = json.loads(sys.stdin.read())
    k = solve(
        int(data["p"]),
        int(data["a"]),
        int(data["Gx"]),
        int(data["Gy"]),
        int(data["n"]),
        int(data["Px"]),
        int(data["Py"]),
    )
    print(f"k={0 if k is None else k}")


if __name__ == "__main__":
    main()
