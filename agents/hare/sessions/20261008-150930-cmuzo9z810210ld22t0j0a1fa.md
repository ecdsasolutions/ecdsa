#!/usr/bin/env python3
"""48-bit ECDLP: negation-map BSGS, packed uint64 table, Montgomery batch inv.

k = i*M + j or i*M - j, with M = 2*m+1, m = isqrt(n)//2+1.
Baby x packed as (fp<<25)|((j<<1)|yParity), fp from a 64-bit mix.
"""
import json
import sys
from array import array
from math import isqrt

GOLD = 0x9E3779B97F4A7C15
MASK64 = (1 << 64) - 1
LOW = (1 << 25) - 1
BATCH = 256


def solve(p, a, Gx, Gy, n, Px, Py):
    if (Px % p) == 0 and (Py % p) == 0:
        return 0

    Gx %= p
    Gy %= p
    Px %= p
    Py %= p

    m = isqrt(n) // 2 + 1
    M = 2 * m + 1
    if m >= (1 << 24):
        m = (1 << 24) - 2
        M = 2 * m + 1

    need = (m * 3) // 2 + 8
    slots = 1
    while slots < need:
        slots <<= 1
    if slots < 16:
        slots = 16
    mask = slots - 1
    table = array("Q", [0]) * slots

    # reusable batch buffers
    dxs = [1] * (BATCH + 1)
    pref = [0] * (BATCH + 1)
    ivs = [0] * (BATCH + 1)

    def inv_into(L):
        acc = dxs[0] % p
        pref[0] = acc
        for i in range(1, L):
            acc = (acc * dxs[i]) % p
            pref[i] = acc
        inv = pow(acc, -1, p)
        for i in range(L - 1, 0, -1):
            ivs[i] = (inv * pref[i - 1]) % p
            inv = (inv * dxs[i]) % p
        ivs[0] = inv

    def insert(x, j, y):
        h = (x * GOLD) & MASK64
        fp = h >> 25
        slot = h & mask
        val = (fp << 25) | ((j << 1) | (y & 1))
        t = table
        while t[slot]:
            slot = (slot + 1) & mask
        t[slot] = val

    def match_k(x, y, i):
        """Return k if x is a baby x (fp match), else 0. Caller verifies."""
        h = (x * GOLD) & MASK64
        fp = h >> 25
        slot = h & mask
        t = table
        ypar = y & 1
        iM = i * M
        while True:
            v = t[slot]
            if v == 0:
                return 0
            if (v >> 25) == fp:
                payload = v & LOW
                j = payload >> 1
                if ypar == (payload & 1):
                    k = iM + j
                else:
                    k = iM - j
                if k <= 0:
                    k += n
                elif k >= n:
                    k -= n
                return k
            slot = (slot + 1) & mask

    # S[i] = i*G for i=1..BATCH
    Sx = [0] * (BATCH + 1)
    Sy = [0] * (BATCH + 1)
    Sx[1] = Gx
    Sy[1] = Gy
    if BATCH >= 2:
        lam = ((3 * Gx * Gx + a) * pow((2 * Gy) % p, -1, p)) % p
        nx = (lam * lam - 2 * Gx) % p
        ny = (lam * (Gx - nx) - Gy) % p
        Sx[2] = nx
        Sy[2] = ny
    for i in range(3, BATCH + 1):
        sx, sy = Sx[i - 1], Sy[i - 1]
        dx = (Gx - sx) % p
        lam = ((Gy - sy) * pow(dx, -1, p)) % p
        nx = (lam * lam - sx - Gx) % p
        ny = (lam * (sx - nx) - sy) % p
        Sx[i] = nx
        Sy[i] = ny

    limit = m if m < BATCH else BATCH
    for i in range(1, limit + 1):
        insert(Sx[i], i, Sy[i])

    rx, ry = Sx[limit], Sy[limit]
    j = limit
    while j < m:
        cnt = BATCH if j + BATCH <= m else m - j
        dbl = 0
        for t in range(cnt):
            dx = (Sx[t + 1] - rx) % p
            if dx == 0:
                dbl |= 1 << t
                dxs[t] = 1
            else:
                dxs[t] = dx
        inv_into(cnt)
        for t in range(cnt):
            sx = Sx[t + 1]
            sy = Sy[t + 1]
            if (dbl >> t) & 1:
                if (sy - ry) % p == 0:
                    lam = ((3 * rx * rx + a) * pow((2 * ry) % p, -1, p)) % p
                    nx = (lam * lam - 2 * rx) % p
                    ny = (lam * (rx - nx) - ry) % p
                else:
                    # R + S = infinity; not expected in baby walk
                    nx, ny = 0, 0
            else:
                lam = ((sy - ry) * ivs[t]) % p
                nx = (lam * lam - rx - sx) % p
                ny = (lam * (rx - nx) - ry) % p
            jj = j + t + 1
            insert(nx, jj, ny)
            if t + 1 == cnt:
                rx, ry = nx, ny
        j += cnt

    # D = M*G via double-and-add
    def add(x1, y1, x2, y2):
        if x1 is None:
            return x2, y2
        if x2 is None:
            return x1, y1
        if x1 == x2:
            if (y1 + y2) % p == 0:
                return None, None
            lam = ((3 * x1 * x1 + a) * pow((2 * y1) % p, -1, p)) % p
            nx = (lam * lam - 2 * x1) % p
            ny = (lam * (x1 - nx) - y1) % p
            return nx, ny
        lam = ((y2 - y1) * pow((x2 - x1) % p, -1, p)) % p
        nx = (lam * lam - x1 - x2) % p
        ny = (lam * (x1 - nx) - y1) % p
        return nx, ny

    def mul(k, x, y):
        rx_ = ry_ = None
        while k:
            if k & 1:
                rx_, ry_ = add(rx_, ry_, x, y)
            x, y = add(x, y, x, y)
            k >>= 1
        return rx_, ry_

    Dx, Dy = mul(M, Gx, Gy)
    # U[i] = -(i*D)
    Ux = [0] * (BATCH + 1)
    Uy = [0] * (BATCH + 1)
    Ux[1] = Dx
    Uy[1] = (-Dy) % p
    if BATCH >= 2:
        lam = ((3 * Ux[1] * Ux[1] + a) * pow((2 * Uy[1]) % p, -1, p)) % p
        nx = (lam * lam - 2 * Ux[1]) % p
        ny = (lam * (Ux[1] - nx) - Uy[1]) % p
        Ux[2] = nx
        Uy[2] = ny
    ux1, uy1 = Ux[1], Uy[1]
    for i in range(3, BATCH + 1):
        sx, sy = Ux[i - 1], Uy[i - 1]
        dx = (ux1 - sx) % p
        if dx == 0:
            lam = ((3 * sx * sx + a) * pow((2 * sy) % p, -1, p)) % p
            nx = (lam * lam - 2 * sx) % p
            ny = (lam * (sx - nx) - sy) % p
        else:
            lam = ((uy1 - sy) * pow(dx, -1, p)) % p
            nx = (lam * lam - sx - ux1) % p
            ny = (lam * (sx - nx) - sy) % p
        Ux[i] = nx
        Uy[i] = ny

    def verify(k):
        x, y = mul(k, Gx, Gy)
        return x == Px and y == Py

    # giant steps: R_i = P - i*D, i = 0,1,2,...
    # k = i*M ± j
    qx, qy = Px, Py
    i_base = 0
    # cover k up to n-1: i*M - m <= n-1 + something
    i_max = (n - 1) // M + 2

    def consider(x, y, i):
        k = match_k(x, y, i)
        if k and verify(k):
            return k
        return 0

    hit = consider(qx, qy, 0)
    if hit:
        return hit

    i_base = 0
    while i_base < i_max:
        cnt = BATCH if i_base + BATCH <= i_max else i_max - i_base
        if cnt <= 0:
            break
        # compute R + U[t] for t=1..cnt from the same R = P - i_base*D
        dbl = 0
        inf = 0
        for t in range(cnt):
            ux = Ux[t + 1]
            dx = (ux - qx) % p
            if dx == 0:
                dbl |= 1 << t
                dxs[t] = 1
                if (Uy[t + 1] + qy) % p == 0:
                    inf |= 1 << t
            else:
                dxs[t] = dx
        inv_into(cnt)
        nxt_x = nxt_y = None
        for t in range(cnt):
            i = i_base + t + 1
            if (inf >> t) & 1:
                # P - i*D = infinity => k = i*M
                k = i * M
                if k >= n:
                    k -= n
                if k and verify(k):
                    return k
                if t + 1 == cnt:
                    # next R is infinity; should not continue usefully
                    nxt_x, nxt_y = None, None
                continue
            ux = Ux[t + 1]
            uy = Uy[t + 1]
            if (dbl >> t) & 1:
                lam = ((3 * qx * qx + a) * pow((2 * qy) % p, -1, p)) % p
                nx = (lam * lam - 2 * qx) % p
                ny = (lam * (qx - nx) - qy) % p
            else:
                lam = ((uy - qy) * ivs[t]) % p
                nx = (lam * lam - qx - ux) % p
                ny = (lam * (qx - nx) - qy) % p
            hit = consider(nx, ny, i)
            if hit:
                return hit
            if t + 1 == cnt:
                nxt_x, nxt_y = nx, ny
        if nxt_x is None:
            break
        qx, qy = nxt_x, nxt_y
        i_base += cnt

    return None


def main():
    data = json.load(sys.stdin)
    k = solve(
        int(data["p"]),
        int(data["a"]),
        int(data["Gx"]),
        int(data["Gy"]),
        int(data["n"]),
        int(data["Px"]),
        int(data["Py"]),
    )
    if k is None:
        print("k=NOTFOUND", file=sys.stderr)
        sys.exit(1)
    print(f"k={k}")


if __name__ == "__main__":
    main()
